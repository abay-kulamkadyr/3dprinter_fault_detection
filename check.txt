#to launch live streaming into nginx server
gst-launch-1.0 -v v4l2src device=/dev/video0 ! image/jpeg,width=1920,height=1080,framerate=30/1 ! jpegdec ! videoconvert ! x264enc tune=zerolatency bitrate=2000 speed-preset=superfast ! flvmux streamable=true ! rtmpsink location='rtmp://localhost/live/stream'

#to install nginx with tcmp support
sudo apt install nginx 
sudo apt install libnginx-mode-rtmp
